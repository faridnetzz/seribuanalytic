[source3]

[streammux]
gpu-id=0
batch-size=8
batched-push-timeout=40000
width=1920
height=1080
live-source=1

# --- PGIE: detektor utama (orang + kendaraan) ---
[primary-gie]
enable=1
gpu-id=0
config-file=config_pgie_detector.txt

# --- Tracker (NvDCF) untuk ID persisten + dwell time parkir ---
[tracker]
enable=1
tracker-width=640
tracker-height=384
ll-lib-file=/opt/nvidia/deepstream/deepstream/lib/libnvds_nvmultiobjecttracker.so
ll-config-file=config_tracker.yml
gpu-id=0

# --- SGIE: Pedestrian Attribute Recognition (11 atribut) ---
# DINONAKTIFKAN default: PAR + Re-ID dijalankan oleh engine Python pada crop
# yang dikirim DeepStream. Aktifkan (enable=1) hanya bila ingin PAR di dalam
# pipeline DeepStream alih-alih di engine.
[secondary-gie0]
enable=0
gpu-id=0
config-file=config_sgie_par.txt
operate-on-gie-id=1
operate-on-class-ids=0      ; class 0 = person

# --- Konversi metadata -> payload JSON ---
[message-converter]
enable=1
msg-conv-config=config_msgconv.txt
# Skema custom (payload SIGAP). 0=full DS schema, 1=minimal, 256+=custom lib
msg-conv-payload-type=1
msg-conv-msg2p-new-api=1

# --- Broker: kirim metadata PERCEPTION ke MQTT (Mosquitto) ---
# Penting: DeepStream menerbitkan output mentah (objek+track) ke
# sigap/perception/*, BUKAN sigap/events/*. Engine Python mengonsumsinya,
# menjalankan model per modul, lalu menerbitkan event final ke sigap/events/*.
[message-broker]
enable=1
proto-lib=/opt/nvidia/deepstream/deepstream/lib/libnvds_mqtt_proto.so
conn-str=mqtt;1883
topic=sigap/perception/objects
# File koneksi tambahan (host broker, qos, dll)
broker-config-file=cfg_mqtt.txt

[sink0]
enable=1
type=1            ; 1 = fakesink (tanpa display di server headless)
sync=0

[osd]
enable=1
display-text=1
display-bbox=1
